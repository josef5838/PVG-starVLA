class TinySurface(torch.nn.Module):
    encode_camera_tokens_separately = True

    def __init__(self):
        super().__init__()
        self.compressor = Compressor(embed_dim=8, depth=1, num_heads=2, num_latent_tokens=128)
        self.token_projector = torch.nn.Linear(8, 8)
        self.field_decoder = torch.nn.Linear(8, 6)
        self.camera_tokens_compressor = torch.nn.Linear(8, 8)

    def _build_aggregated_tokens(self, token_list, patch_start_idx, **kwargs):
        return self.token_projector(token_list[0].flatten(1, 2)), None

    def get_compressed_tokens(self, token_list, patch_start_idx, **kwargs):
        tokens, mask = self._build_aggregated_tokens(token_list, patch_start_idx, **kwargs)
        return self.compressor(tokens, read_ca_mask=mask)


class TinyBackbone(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.vggt = TinyVGGT(torch.nn.Linear(3, 8))
        self.surface_net = TinySurface()


def make_policy(condition_state=False):
    cfg = OmegaConf.create({'framework': {
        'surflo': {'train_compressor': True, 'compressor_state_conditioning': condition_state,
                   'append_camera_token': False, 'append_per_view_camera_tokens': False,
                   'image_size': 14},
        'action_model': {'state_dim': 14, 'action_hidden_dim': 8, 'num_condition_tokens': 132},
        'state_conditioning': 'parts', 'state_parts_layout': 'robotwin',
    }, 'trainer': {'freeze_modules': '', 'learning_rate': {'base': .001,
                   'surflo_backbone.surface_net.compressor': .0001}}})
    with patch.object(SurfloDiT, '_load_surflo', return_value=TinyBackbone()), \
         patch('starVLA.model.framework.VLM4A.SurfloDiT.get_action_model',
               return_value=torch.nn.Linear(8, 8)):
        model = SurfloDiT(cfg)

    return model


class TinyVGGT(torch.nn.Module):
    def __init__(self, projection):
        super().__init__()
        self.projection = projection

    def features(self, images):
        assert not torch.is_grad_enabled()
        assert not self.training
        return self.projection(images.mean((-1, -2))).unsqueeze(2)

    def forward(self, images, return_aggregated_tokens):
        return {'aggregated_tokens_list': [self.features(images)], 'patch_start_idx': 0,
                'world_points': torch.zeros(len(images), 3, 1, 1, 3)}


def examples():
    return [{'image': [torch.rand(3, 14, 14) for _ in range(3)],
             'state': np.linspace(-1, 1, 14, dtype=np.float32)} for _ in range(2)]


def check_gradients_and_resume(condition_state, tmp_path):
    torch.manual_seed(42)
    model = make_policy(condition_state)
    model.train()
    cfg = OmegaConf.create({'trainer': {'freeze_modules': '', 'learning_rate': {
        'base': .001, 'surflo_backbone.surface_net.compressor': .0001}}})
    groups = build_param_lr_groups(model, cfg)
    optimizer = torch.optim.AdamW(groups)
    assert all(p.requires_grad for group in groups for p in group['params'])
    before = model.surflo_backbone.surface_net.compressor.latent_tokens.detach().clone()
    frozen = {n: p.detach().clone() for n, p in model.surflo_backbone.named_parameters()
              if not p.requires_grad}
    batch = examples()
    model._condition(batch, apply_state_dropout=False).square().mean().backward()
    assert model.surflo_backbone.surface_net.compressor.latent_tokens.grad.abs().sum() > 0
    assert not model.surflo_backbone.vggt.training
    assert not model.surflo_backbone.surface_net.field_decoder.training
    if condition_state:
        assert model.compressor_state_conditioner.state_encoder[-1].weight.grad.abs().sum() > 0
    optimizer.step()
    assert not torch.equal(before, model.surflo_backbone.surface_net.compressor.latent_tokens)
    for n, p in model.surflo_backbone.named_parameters():
        if n in frozen:
            assert p.grad is None
            torch.testing.assert_close(p, frozen[n], rtol=0, atol=0)
    path = tmp_path / 'checkpoint.pt'
    torch.save({'model': model.state_dict(), 'optimizer': optimizer.state_dict()}, path)
    restored = make_policy(condition_state)
    restored_optimizer = torch.optim.AdamW(build_param_lr_groups(restored, cfg))
    checkpoint = torch.load(path, weights_only=True)
    restored.load_pretrained_state_dict(checkpoint['model'])
    restored_optimizer.load_state_dict(checkpoint['optimizer'])
    model.eval()
    restored.eval()
    torch.testing.assert_close(restored._condition(batch), model._condition(batch), rtol=0, atol=0)
    assert restored_optimizer.state_dict()['state']


def check_zero_initialization_and_state_sensitivity():
    torch.manual_seed(7)
    compressor = Compressor(embed_dim=8, depth=1, num_heads=2, num_latent_tokens=4)
    conditioner = SurfloStateQueryConditioner(14, 8)
    tokens = torch.randn(2, 6, 8)
    state = torch.randn(2, 14)
    reference, _ = compressor(tokens)
    actual, _ = conditioner(compressor, tokens, state)
    torch.testing.assert_close(actual, reference, rtol=0, atol=0)
    torch.nn.init.normal_(conditioner.state_encoder[-1].weight, std=.1)
    torch.nn.init.constant_(conditioner.state_encoder[-1].bias, .3)
    first, _ = conditioner(compressor, tokens, state)
    second, _ = conditioner(compressor, tokens, state + 1)
    assert not torch.allclose(first, second)
    dropped, _ = conditioner(compressor, tokens, state, keep_state=torch.zeros(2))
    torch.testing.assert_close(dropped, reference, rtol=0, atol=0)


def check_state_dropout():
    model = make_policy(True).train()
    model.config.framework.state_dropout_prob = 1.
    torch.nn.init.normal_(model.compressor_state_conditioner.state_encoder[-1].weight)
    torch.nn.init.ones_(model.compressor_state_conditioner.state_encoder[-1].bias)
    batch = examples()
    first = model._condition(batch, apply_state_dropout=True)
    for example in batch:
        example['state'] += 10
    second = model._condition(batch, apply_state_dropout=True)
    torch.testing.assert_close(first, second, rtol=0, atol=0)
    assert not first[:, -4:].count_nonzero()


def check_shared_initialization():
    torch.manual_seed(42)
    plain = make_policy(False)
    plain_rng = torch.get_rng_state()
    torch.manual_seed(42)
    conditioned = make_policy(True)
    torch.testing.assert_close(torch.get_rng_state(), plain_rng, rtol=0, atol=0)
    for name, value in plain.state_dict().items():
        torch.testing.assert_close(value, conditioned.state_dict()[name], rtol=0, atol=0)


class CompressorTrainingTests(unittest.TestCase):
    def test_updated_compressor_weights_cannot_be_silently_omitted(self):
        model = make_policy(False)
        incomplete = {name: value for name, value in model.state_dict().items()
                      if not name.startswith('surflo_backbone.')}
        with self.assertRaisesRegex(RuntimeError, 'updated Surflo weights'):
            model.load_pretrained_state_dict(incomplete)

    def test_plain_gradients_and_checkpoint_roundtrip(self):
        with tempfile.TemporaryDirectory() as directory:
            check_gradients_and_resume(False, Path(directory))

    def test_conditioned_gradients_and_checkpoint_roundtrip(self):
        with tempfile.TemporaryDirectory() as directory:
            check_gradients_and_resume(True, Path(directory))

    def test_zero_initialization_and_state_sensitivity(self):
        check_zero_initialization_and_state_sensitivity()

    def test_state_dropout(self):
        check_state_dropout()

    def test_shared_initialization(self):
        check_shared_initialization()


if __name__ == '__main__':
    unittest.main()
