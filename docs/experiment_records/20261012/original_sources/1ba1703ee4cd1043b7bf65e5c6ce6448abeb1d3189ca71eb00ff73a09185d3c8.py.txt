"""Read progress for a launched independent-camera conditioning experiment."""
import argparse
import json
from pathlib import Path


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--manifest',type=Path,required=True)
    args=parser.parse_args();manifest=json.loads(args.manifest.read_text());rows=[]
    sample_orders={}
    for run in manifest['runs']:
        progress=Path(run['output_dir'])/'stream_training_status.json'
        row=json.loads(progress.read_text()) if progress.exists() else dict(status=run.get('status','prepared'))
        process=Path('/proc')/str(run.get('pid'))/'cmdline'
        row['process_alive']=bool(process.exists() and str(run['config']).encode() in process.read_bytes())
        row.update(variant=run['variant'],gpu=run['gpu'],pid=run['pid'],log=run['log'],output_dir=run['output_dir'])
        rows.append(row)
        order_path=Path(run['output_dir'])/'initial_sample_order.jsonl'
        if order_path.exists():
            order=[json.loads(x) for x in order_path.read_text().splitlines() if x]
            if len(order)>=16:sample_orders[run['variant']]=order[:16]
    order_audit=dict(compared_variants=list(sample_orders),
        matched=len(sample_orders)>=2 and all(order==next(iter(sample_orders.values())) for order in sample_orders.values()),
        all_variants_available=len(sample_orders)==len(manifest['runs']))
    print(json.dumps(dict(manifest=str(args.manifest.resolve()),status=manifest['status'],
        queue_workers=manifest.get('queue_workers',[]),initial_sample_order_audit=order_audit,runs=rows),indent=2))


if __name__=='__main__':main()
