"""Three independent frozen Omega history calls; camera/register-only action memory."""
from contextlib import nullcontext

import numpy as np
import torch
from torch import nn

from starVLA.model.framework.VLM4A.Omega512DiT import Omega512DiT, STATE_PARTS
from starVLA.model.modules.projector.omega_stream_fusion import OmegaStreamFusion
from starVLA.model.tools import FRAMEWORK_REGISTRY


@FRAMEWORK_REGISTRY.register('OmegaCameraStreamDiT')
class OmegaCameraStreamDiT(Omega512DiT):
    def __init__(self, config=None, **kwargs):
        super().__init__(config, **kwargs)
        cfg = self.config.framework.streams
        if int(self.config.framework.omega.train_last_blocks) != 0 or self.readout_type != 'dense':
            raise ValueError('Camera-stream experiments require a frozen Omega and no dense readout adapter')
        if bool(cfg.temporal_rope):
            raise ValueError('Temporal RoPE is intentionally excluded from this protocol')
        if int(cfg.history_length) != 8 or str(cfg.conditioning) not in {'direct', 'fused'}:
            raise ValueError('Expected eight-image histories and direct/fused conditioning')
        self.history_length = int(cfg.history_length)
        self.conditioning = str(cfg.conditioning)
        width = int(cfg.width)
        decoder_width = int(self.config.framework.action_model.action_hidden_dim)
        # Preserve common decoder initialization and RNG across ablations.
        with torch.random.fork_rng(devices=[]):
            self.visual_proj = nn.Linear(2048, width)
            self.position_proj = nn.Identity()
            self.view_embeddings = nn.Parameter(torch.randn(3, width)*.02)
            self.time_embeddings = nn.Parameter(torch.randn(8, width)*.02)
            self.token_type_embeddings = nn.Parameter(torch.randn(2, width)*.02)
            self.memory_proj = nn.Linear(width, decoder_width)
            self.memory_output_norm = nn.LayerNorm(decoder_width)
            if self.conditioning == 'fused':
                self.visual_readout = OmegaStreamFusion(width, int(cfg.depth), int(cfg.heads), int(cfg.output_queries))

    def _history_tokens(self, examples):
        rows = []
        for row in examples:
            history = row.get('image_history')
            if history is None or len(history) != 3 or any(len(view) != 8 for view in history):
                raise ValueError('Expected image_history[camera][time]: three chronological eight-image streams')
            # Reuse official-equivalent preprocessing from the parent, time-major.
            rows.extend(dict(image=[history[c][t] for c in range(3)]) for t in range(8))
        prepared = super()._prepare_images(rows)
        images = prepared.reshape(len(examples), 8, 3, 3, *prepared.shape[-2:])
        images = images.transpose(1, 2)  # [B, physical camera, history, RGB, H, W]
        device = self.visual_proj.weight.device
        amp = torch.autocast('cuda', dtype=torch.bfloat16) if device.type == 'cuda' else nullcontext()
        outputs = []
        with torch.no_grad(), amp:
            for camera in range(3):
                # A separate call per physical camera; histories from batch elements never mix.
                layers, patch_start = self.omega.aggregator(images[:, camera])
                if patch_start != 17:
                    raise ValueError('Expected one camera token and sixteen registers')
                features = layers[int(self.config.framework.omega.feature_layer)]
                if features is None:
                    raise ValueError('Requested Omega layer is not cached')
                outputs.append(features[:, :, :17].detach().contiguous())
                del layers, features
        self._stream_input_shape = list(images.shape)
        return torch.stack(outputs, dim=1)

    def _condition(self, examples, apply_dropout=False, state_conditioning=None, visual_conditioning=None):
        state_mode = self.config.framework.state_conditioning if state_conditioning is None else state_conditioning
        if state_mode not in {'full', 'none', 'decoder_only'}:
            raise ValueError('Camera-stream readout does not consume robot state; use full/decoder_only/none')
        use_visual = self.config.framework.visual_conditioning if visual_conditioning is None else visual_conditioning
        device, dtype = self.visual_proj.weight.device, self.visual_proj.weight.dtype
        state = torch.as_tensor(np.stack([np.asarray(row['state'], dtype=np.float32).reshape(-1)
                                         for row in examples]), device=device, dtype=dtype)
        if state.shape != (len(examples), 14) or not torch.isfinite(state).all():
            raise ValueError('Expected finite normalized RoboTwin state [B,14]')
        tasks = [self._task_key(row.get('lang', row.get('instruction'))) for row in examples]
        if set(tasks)-set(self.task_names):
            raise ValueError(f'Unknown task instruction: {tasks}')
        task = self.task_embedding(torch.tensor([self.task_names.index(name) for name in tasks], device=device))
        keep_state = torch.ones(len(examples), device=device, dtype=dtype)
        if apply_dropout:
            keep_state = (torch.rand(len(examples), device=device) >= self.config.framework.state_dropout_prob).to(dtype)
        if state_mode == 'none':
            keep_state = keep_state*0
        parts = torch.stack([projection(state[:, part]) for projection, part in zip(self.state_proj, STATE_PARTS)], dim=1)
        parts = (parts+self.state_part_embeddings)*keep_state[:,None,None]*self.config.framework.state_scale
        condition = torch.cat([parts, task[:,None]], dim=1)
        raw = self._history_tokens(examples)
        features = self.visual_proj(self.visual_norm(raw.to(dtype)))
        kinds = torch.tensor([0]+[1]*16, device=device)
        features = (features+self.view_embeddings[None,:,None,None]
                    +self.time_embeddings[None,None,:,None]
                    +self.token_type_embeddings[kinds][None,None,None])
        memory = features.flatten(1, 3)
        if self.visual_readout is not None:
            memory = self.visual_readout(memory)
        memory = self.memory_output_norm(self.memory_proj(memory))
        if not use_visual:
            memory = memory*0
        if apply_dropout:
            keep = (torch.rand(len(examples), device=device) >= self.config.framework.condition_dropout_prob).to(dtype)
            condition = condition*keep[:,None,None]; memory = memory*keep[:,None,None]
        self.last_visual_info = dict(input_shape=self._stream_input_shape, calls_per_batch=3,
            images_per_call=8, raw_token_shape=list(raw.shape), raw_memory_tokens=408,
            memory_tokens=int(memory.shape[1]), conditioning=self.conditioning,
            patch_tokens_in_action_memory=0, temporal_embedding='learned', temporal_rope=False)
        return condition, memory
